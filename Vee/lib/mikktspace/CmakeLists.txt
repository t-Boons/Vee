cmake_minimum_required(VERSION 3.16)
project(mikktspace LANGUAGES C)

add_library(mikktspace STATIC
    "${CMAKE_CURRENT_LIST_DIR}/mikktspace.c"
)

set_source_files_properties("${CMAKE_CURRENT_LIST_DIR}/mikktspace.c" PROPERTIES LANGUAGE C)

target_include_directories(mikktspace PUBLIC "${CMAKE_CURRENT_LIST_DIR}")

set_target_properties(mikktspace PROPERTIES
    C_STANDARD 99
    C_STANDARD_REQUIRED ON
)
